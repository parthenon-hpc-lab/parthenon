# tests/CMakeLists.txt

# We assume BUILD_TESTING and enable_testing() are already set at the top level
# via include(CTest)

include(FetchContent)

# Declare and fetch Catch2 (v3)
FetchContent_Declare(
    Catch2
    GIT_REPOSITORY https://github.com/catchorg/Catch2.git
    GIT_TAG        v3.5.3   # or a stable version you prefer
)

FetchContent_MakeAvailable(Catch2)

# Bring in the Catch2 CMake helper functions (includes catch_discover_tests)
list(APPEND CMAKE_MODULE_PATH "${Catch2_SOURCE_DIR}/extras")
include(Catch)

set(TESTS_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

# Define the test executable
add_executable(malg_tests
    "${TESTS_DIR}/catch2_define.cpp"  
    "${TESTS_DIR}/implicit_qr_tests.cpp"  # add more test files here as needed
    "${TESTS_DIR}/svd_tests.cpp"  # add more test files here as needed
)

target_link_libraries(malg_tests
    PRIVATE
        MALG
        Catch2::Catch2WithMain
)

# Let Catch2 register individual test cases with CTest
catch_discover_tests(malg_tests)
